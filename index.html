// ID da planilha padrão: usada quando o app não enviar nenhuma planilha. var SPREADSHEET_ID = ''; // Abre a planilha pelo ID recebido; se vier vazio, usa a planilha padrão. function getSheet(planilhaId, sheetName) { var id = String(planilhaId || '').trim(); var m = id.match(/\/d\/([a-zA-Z0-9_-]+)/); if (m) id = m[1]; var ss = SpreadsheetApp.openById(id || SPREADSHEET_ID); var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); } return sheet; } // Envia os dados mais recentes da planilha para o aplicativo function doGet(e) { try { if (e && e.parameter && e.parameter.acao === 'usuario') { return responderUsuario_(e.parameter.email); } var planilha = (e && e.parameter) ? e.parameter.planilha : ''; var sheetShortcuts = getSheet(planilha, "Atalhos"); var lastRow = sheetShortcuts.getLastRow(); var lastCol = sheetShortcuts.getLastColumn(); var shortcuts = []; if (lastRow > 1 && lastCol >= 1) { var colsToRead = Math.max(lastCol, 11); var values = sheetShortcuts.getRange(2, 1, lastRow - 1, colsToRead).getValues(); values.forEach(function(row) { if (row[0] || row[1]) { shortcuts.push({ id: String(row[0]), title: String(row[1]), name: String(row[1]), file: String(row[2]), category: String(row[3] || 'Geral'), isFavorite: String(row[4]).toUpperCase() === 'SIM', customIcon: String(row[5] || ''), visitCount: Number(row[6]) || 0, showDesktop: row[7] !== '' ? String(row[7]).toUpperCase() !== 'NAO' : true, showMobile: row[8] !== '' ? String(row[8]).toUpperCase() !== 'NAO' : true, subcategory: String(row[9] || ''), subSubcategory: String(row[10] || '') }); } }); } var sheetCategories = getSheet(planilha, "Categorias"); var lastCatRow = sheetCategories.getLastRow(); var categories = {}; if (lastCatRow > 1) { var catValues = sheetCategories.getRange(2, 1, lastCatRow - 1, 2).getValues(); catValues.forEach(function(row) { var catName = String(row[0] || '').trim(); var catDataStr = String(row[1] || '').trim(); if (catName && catDataStr) { try { categories[catName] = JSON.parse(catDataStr); } catch (err) { categories[catName] = { icon: catDataStr }; } } }); } return ContentService.createTextOutput(JSON.stringify({ status: "success", shortcuts: shortcuts, categories: categories })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // Recebe os dados do aplicativo e salva na planilha em lote function doPost(e) { try { var data = JSON.parse(e.postData.contents); if (data.acao === 'admin') return responderAdmin_(data); var planilha = data.planilha || ''; var sheetShortcuts = getSheet(planilha, "Atalhos"); var rows = [['ID', 'Título', 'Link', 'Categoria', 'Favorito', 'IconeCustom', 'Visitas', 'Computador', 'Celular', 'Subcategoria', 'SubSubcategoria']]; if (Array.isArray(data.shortcuts)) { data.shortcuts.forEach(function(s) { rows.push([ s.id || '', s.title || s.name || '', s.file || '', s.category || 'Geral', s.isFavorite ? 'SIM' : 'NAO', s.customIcon || '', s.visitCount || 0, s.showDesktop !== false ? 'SIM' : 'NAO', s.showMobile !== false ? 'SIM' : 'NAO', s.subcategory || '', s.subSubcategory || '' ]); }); } sheetShortcuts.clearContents(); sheetShortcuts.clearFormats(); if (rows.length > 0) { sheetShortcuts.getRange(1, 1, rows.length, rows[0].length).setValues(rows); } var sheetCategories = getSheet(planilha, "Categorias"); var catRows = [['Categoria', 'Dados']]; if (data.categories && typeof data.categories === 'object') { for (var catName in data.categories) { var catVal = data.categories[catName]; var valStr = (typeof catVal === 'object') ? JSON.stringify(catVal) : String(catVal); catRows.push([catName, valStr]); } } sheetCategories.clearContents(); sheetCategories.clearFormats(); if (catRows.length > 0) { sheetCategories.getRange(1, 1, catRows.length, catRows[0].length).setValues(catRows); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
